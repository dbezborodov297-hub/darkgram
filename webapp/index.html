async function doSearch(){ const inp = document.getElementById('searchInput'); const query = inp.value.trim(); if (!query || searchInProgress) return; if (player.tokens < 1){ toast('Недостаточно токенов', true); haptic('error'); return; } searchInProgress = true; document.getElementById('searchBtn').disabled = true; document.getElementById('searchBtnText').textContent = 'Поиск...'; document.getElementById('resultBox').classList.remove('show'); document.getElementById('emptyState').style.display = 'none'; document.getElementById('searchingBox').classList.add('show'); document.getElementById('answerText').innerHTML = ''; startTime = Date.now(); const circle = document.getElementById('timerCircle'); const offset = 251.3 * Math.min(1, elapsed / 30); circle.setAttribute('stroke-dashoffset', offset); }, 100); try { const response = await fetch('/api/search/stream', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({uid: UID, query: query}) }); const reader = response.body.getReader(); const decoder = new TextDecoder(); let buffer = ''; let fullText = ''; let finalData = null; while (true) { const {done, value} = await reader.read(); if (done) break; buffer += decoder.decode(value, {stream: true}); const lines = buffer.split('\n\n'); buffer = lines.pop(); for (const line of lines) { if (!line.startsWith('data: ')) continue; const data = JSON.parse(line.slice(6)); if (data.type === 'phase'){ document.getElementById('searchingStatus').textContent = data.text; currentPhase = data.phase - 1; updatePhase(currentPhase); } else if (data.type === 'chunk'){ fullText += data.text; // Показываем текст постепенно document.getElementById('searchingBox').classList.remove('show'); document.getElementById('resultBox').classList.add('show'); document.getElementById('emptyState').style.display = 'none'; document.getElementById('answerText').innerHTML = fullText + '▌'; } else if (data.type === 'done'){ finalData = data; } else if (data.type === 'error'){ toast(data.message, true); } } } clearInterval(timerInterval); document.getElementById('searchingBox').classList.remove('show'); if (finalData){ // Финальный ответ без курсора lastAnswer = fullText; document.getElementById('answerText').innerHTML = fullText; document.getElementById('copyBtn').classList.remove('copied'); // Источники let sourcesHtml = ''; if (finalData.sources && finalData.sources.length){ sourcesHtml = '
' + '' + 'Источники (' + finalData.sources.length + ')
' + finalData.sources.map((s, i) => '' + '
' + (i+1) + '
' + '
' + '
' + esc(s.title) + '
' + '
' + esc(s.url) + '
' + '
' + '' ).join(''); } document.getElementById('sourcesBlock').innerHTML = sourcesHtml; document.getElementById('resultMeta').innerHTML = '
Поиск: ' + finalData.search_time + 'с
' + '
AI: ' + finalData.ai_time + 'с
' + '
Всего: ' + finalData.total_time + 'с
' + '
-' + finalData.cost + ' токен
'; player.tokens = finalData.tokens_left; player.total_requests = (player.total_requests || 0) + 1; player.total_spent = (player.total_spent || 0) + finalData.cost; document.getElementById('tokensHeader').textContent = player.tokens; renderProfile(); loadHistory(); loadStats(); haptic('success'); } } catch(e){ clearInterval(timerInterval); document.getElementById('searchingBox').classList.remove('show'); toast('Ошибка. Попробуй ещё раз.', true); document.getElementById('emptyState').style.display = 'block'; } searchInProgress = false; document.getElementById('searchBtn').disabled = false; document.getElementById('searchBtnText').textContent = 'Найти'; }
